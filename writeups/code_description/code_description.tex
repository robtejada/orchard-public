Blue boxes show the initialization phase; cyan dotted boxes indicate
user-configurable selections (planet type, EOS, and atmosphere boundary
condition); orange boxes denote the main computational steps within each
timestep; yellow diamonds are decision points for state validation and
error-controlled step acceptance; the violet box marks data output; and
green dashed boxes indicate auxiliary modules called by the core routines.

and metal fraction, computed over zones above a configurable pressure
floor.}

% ─────────────────────────────────────────────────────────────────────────────
% ALGORITHM: ORCHARD Evolution Pseudocode
% ─────────────────────────────────────────────────────────────────────────────

\begin{algorithm*}[t!]
\DontPrintSemicolon
\SetAlgoLined
\SetKwInput{KwInput}{Input}
\SetKwInput{KwOutput}{Output}
\SetKwFunction{InitGrid}{InitializeGrid}
\SetKwFunction{HSE}{HydrostaticEquilibrium}
\SetKwFunction{UpdateSYZ}{UpdateSYZ}
\SetKwFunction{AtmBC}{AtmosphereBC}
\SetKwFunction{SaveProfiles}{SaveProfiles}

\KwInput{$M_p$, $S_0$, $Y_0$, $Z_0$, $M_{\rm c}$, $M_{\rm Fe}$, $t_{\rm final}$, $\Delta t_0$, $\epsilon$, $\Delta t_{\rm max}$}
\KwOutput{$r(m,t)$, $P(m,t)$, $\rho(m,t)$, $T(m,t)$, $S(m,t)$, $Y(m,t)$, $Z(m,t)$}

\BlankLine
\tcc{Phase I: Initialization (\texttt{initial.py})}
$(r_b,\, P,\, m_b,\, \Delta m,\, S,\, Y,\, Z,\, f_{\rm rock},\, k_{\rm c},\, k_{\rm Fe}) \leftarrow$ \InitGrid{$N,\, M_p,\, S_0,\, Y_0,\, Z_0$}\;
Instantiate EOS: \texttt{mixtures\_eos} (H/He/Z),\; \texttt{mantle\_eos} (rock),\; \texttt{core\_eos} (Fe)\;
$(r_b,\, P,\, \rho,\, T,\, g,\, \omega,\, J_{2n}) \leftarrow$ \HSE{$S,\, r_b,\, P,\, m_b,\, Y,\, Z,\, f_{\rm rock},\, k_{\rm c},\, k_{\rm Fe}$}\;
$(T_{\rm int},\, T_{\rm eff}) \leftarrow$ \AtmBC{$S,\, Y,\, Z,\, g$}\;
$t \leftarrow 0$;\quad $\Delta t \leftarrow \Delta t_0$;\quad $n_{\rm retry} \leftarrow 0$\;

\BlankLine
\tcc{Phase II: Main Evolution Loop (\texttt{evolution.py})}
\While{$t \leq t_{\rm final}$}{
    \BlankLine
    \tcc{Step 1: Transport solve --- global Newton--Raphson, $3N$ unknowns (\texttt{transport.py})}
    $(S_{\rm new},\, Y_{\rm new},\, Z_{\rm new},\, \mathcal{F},\, n_{\rm itr}) \leftarrow$ \UpdateSYZ{$T,\, r_b,\, P,\, \rho,\, m_b,\, \Delta m,\, T_{\rm int},\, \Delta t,\, Y,\, Z,\, S$}\;
    \quad Compute convective, radiative, conductive, semiconvective fluxes\;
    \quad Apply He/Z rain via miscibility module (optional)\;

    \BlankLine
    \tcc{Step 2: Hydrostatic equilibrium --- Henyey relaxation (\texttt{hydrostatic.py})}
    $(r_b^{\,\prime},\, P',\, \rho',\, T',\, g,\, \omega,\, J_{2n}) \leftarrow$ \HSE{$S_{\rm new},\, r_b,\, P,\, m_b,\, Y_{\rm new},\, Z_{\rm new}$}\;
    \quad EOS inversion: $(\rho,\, T) \leftarrow \mathrm{EOS}(S,\, P,\, Y,\, Z)$;\quad optional TOF $\rightarrow J_2,\, J_4,\, J_6,\, J_8$\;

    \BlankLine
    \tcc{Step 3: Validate state}
    \If{\textnormal{NaN detected} \textbf{or} $P$ \textnormal{non-monotonic} \textbf{or} $\max|\Delta\chi| > \Delta\chi_{\rm max}$}{
        $\Delta t \leftarrow \Delta t / 2$;\quad $n_{\rm retry} \leftarrow n_{\rm retry} + 1$;\quad \textbf{continue}\;
    }

    \BlankLine
    \tcc{Step 4: Compute error metric (pressure-masked)}
    $\Delta_e \leftarrow \displaystyle\max_{P_j > P_{\rm floor}}\!\left(\left|\frac{\delta S_j}{S_j}\right|,\; \left|\frac{\delta Y_j}{Y_j}\right|,\; \left|\frac{\delta Z_j}{Z_j}\right|,\; \left|\frac{\delta T_j}{T_j}\right|,\; \left|\frac{\delta \rho_j}{\rho_j}\right|\right)$\;
    $\epsilon_{\rm smooth} \leftarrow (5 - \epsilon)\,\max\!\Big(1 - \big(t \,/\, 0.1\;\mathrm{Gyr}\big)^2,\; 0\Big) + \epsilon$\;

    \BlankLine
    \tcc{Step 5: Accept or reject timestep}
    \eIf{$\Delta_e > \epsilon_{\rm smooth} / 100$}{
        $\Delta t \leftarrow \Delta t / 2$;\quad $n_{\rm retry} \leftarrow n_{\rm retry} + 1$;\quad \textbf{continue}\;
    }{
        \tcc{Step 6: Atmosphere boundary condition (\texttt{atm\_bc.py})}
        Sample $\bar{S},\, \bar{Y},\, \bar{Z}$ at $P \in [10,\,30]$\;bar\;
        $(T_{\rm int},\, T_{\rm eff}) \leftarrow$ \AtmBC{$\bar{S},\, \bar{Y},\, \bar{Z},\, g$}\;

        \BlankLine
        \tcc{Adaptive timestep control}
        $f_{\rm cap} \leftarrow \begin{cases} 1.25 & \text{if } n_{\rm retry} > 0 \\ 2.0 & \text{otherwise}\end{cases}$\;
        $f_{\rm err} \leftarrow \min\!\Big(\epsilon_{\rm smooth}\, /\, \big(100\,\Delta_e\big),\;\; f_{\rm cap}\Big)$\;
        $f_{\rm itr} \leftarrow \begin{cases} (n_{\rm max} - n_{\rm itr} - 1)\,/\,n_{\rm max} + 0.6 & \text{high-precision mode} \\ 1 & \text{otherwise}\end{cases}$\;
        $\Delta t_{\rm new} \leftarrow \min\!\big(f_{\rm err}\cdot f_{\rm itr}\cdot \Delta t,\;\; \Delta t_{\rm max}\big)$\;

        \BlankLine
        \tcc{Step 7: Save and advance}
        \SaveProfiles{$r_b^{\,\prime},\, P',\, \rho',\, T',\, S_{\rm new},\, Y_{\rm new},\, Z_{\rm new}$} to HDF5 or text\;
        $(S,\, Y,\, Z,\, T,\, P,\, \rho,\, r_b) \leftarrow (S_{\rm new},\, Y_{\rm new},\, Z_{\rm new},\, T',\, P',\, \rho',\, r_b^{\,\prime})$\;
        $t \leftarrow t + \Delta t$;\quad $\Delta t \leftarrow \Delta t_{\rm new}$;\quad $n_{\rm retry} \leftarrow 0$\;
    }
}
\caption{The \textsc{orchard} planetary thermal evolution algorithm. The main
loop couples a global Newton--Raphson transport solve for entropy $S$,
helium fraction $Y$, and metal fraction $Z$ (\texttt{transport.py}) with
a Henyey hydrostatic equilibrium solver (\texttt{hydrostatic.py}), using
adaptive error-controlled timestepping with step rejection. Auxiliary
modules provide the equation of state, He/Z miscibility curves,
thermal conductivities, rotational corrections (Theory of Figures),
and atmosphere boundary conditions.}
\label{alg:evolution}
\end{algorithm*}
